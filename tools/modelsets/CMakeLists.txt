cmake_minimum_required(VERSION 3.22)
project(XboxAssetConverter LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set(LIBRW_PLATFORM NULL CACHE STRING "" FORCE)
set(LIBRW_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../librw librw)
add_executable(xbox-convert xbox-convert.cpp vendor/lzokay/lzokay.cpp)
target_link_libraries(xbox-convert PRIVATE librw::librw)
enable_testing()
add_test(NAME xbox-lzo-block-validation COMMAND xbox-convert --self-test)
