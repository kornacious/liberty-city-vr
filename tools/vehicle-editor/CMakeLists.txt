cmake_minimum_required(VERSION 3.22)
project(LibertyVehicleEditor VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
    set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CTest)
add_executable(placement-tests tests.cpp)
target_include_directories(placement-tests PRIVATE ../../overlay/src/vr)
add_test(NAME placement-core COMMAND placement-tests)
add_executable(speedometer-tests speedometer-tests.cpp)
target_include_directories(speedometer-tests PRIVATE ../../overlay/src/vr)
add_test(NAME speedometer COMMAND speedometer-tests)
add_executable(placement-process-tests process-tests.cpp)
if(MSVC)
    target_sources(placement-process-tests PRIVATE utf8.manifest)
endif()
add_test(NAME placement-helper-process COMMAND placement-process-tests)
set(driving_source "${CMAKE_CURRENT_SOURCE_DIR}/../../overlay/src/vr/QuestDrivingVR.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${driving_source}")
file(READ "${driving_source}" driving)
string(FIND "${driving}" "\nvoid\nApplyQuestVehicleViewOffset(" seat_start)
string(FIND "${driving}" "\nvoid\nApplyQuestBikeHorizonLock(" seat_end)
if(seat_start LESS 0 OR seat_end LESS seat_start)
    message(FATAL_ERROR "Cannot locate production seat offset function")
endif()
math(EXPR seat_length "${seat_end}-${seat_start}")
string(SUBSTRING "${driving}" ${seat_start} ${seat_length} seat_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/runtime-seat.inc" "${seat_source}")
string(FIND "${driving}" "\nstatic void\nRefreshViewPlacement(" calibration_start)
string(FIND "${driving}" "\nstatic VehicleViewCalibration *\nGetViewCalibration(" calibration_end)
if(calibration_start LESS 0 OR calibration_end LESS calibration_start)
    message(FATAL_ERROR "Cannot locate production calibration refresh function")
endif()
math(EXPR calibration_length "${calibration_end}-${calibration_start}")
string(SUBSTRING "${driving}" ${calibration_start} ${calibration_length} calibration_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/runtime-calibration.inc" "${calibration_source}")
add_executable(placement-runtime-tests runtime-tests.cpp)
target_include_directories(placement-runtime-tests PRIVATE ../../overlay/src/vr "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME placement-production-seat COMMAND placement-runtime-tests)
set(VEHICLE_EDITOR_MANAGER_RUNTIME "" CACHE PATH "Directory containing the integrated LibertyManager runtime")
if(VEHICLE_EDITOR_MANAGER_RUNTIME)
    if(WIN32)
        set(manager_executable "${VEHICLE_EDITOR_MANAGER_RUNTIME}/LibertyManager.exe")
    else()
        set(manager_executable "${VEHICLE_EDITOR_MANAGER_RUNTIME}/LibertyManager")
    endif()
    add_test(NAME placement-device-bridge COMMAND "${manager_executable}" --tool tools/vehicle-editor/test_device_bridge.py)
endif()
option(VEHICLE_EDITOR_HEADLESS "Build only portable data/placement tests" OFF)
if(NOT VEHICLE_EDITOR_HEADLESS)
    set(LIBRW_PLATFORM GL3 CACHE STRING "" FORCE)
    set(LIBRW_GL3_GFXLIB GLFW CACHE STRING "" FORCE)
    set(LIBRW_TOOLS ON CACHE BOOL "" FORCE)
    # Use an installed GLFW, or build a pinned source dependency.
    find_package(glfw3 QUIET)
    if(NOT TARGET glfw)
        include(FetchContent)
        set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
        set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
        set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.3.8.tar.gz
            URL_HASH SHA256=f30f42e05f11e5fc62483e513b0488d5bceeab7d9c5da0ffe2252ad81816c713
            TIMEOUT 60)
        FetchContent_MakeAvailable(glfw)
        # librw calls find_package again; supply a target-only package shim.
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/glfw-package/glfw3Config.cmake" "set(glfw3_FOUND TRUE)\n")
        set(glfw3_DIR "${CMAKE_CURRENT_BINARY_DIR}/glfw-package")
    endif()
    add_subdirectory(../../librw librw EXCLUDE_FROM_ALL)
    add_executable(liberty-vehicle-editor main.cpp)
    target_include_directories(liberty-vehicle-editor PRIVATE ../../overlay/src/vr)
    target_link_libraries(liberty-vehicle-editor PRIVATE librw_skeleton_imgui)
    if(WIN32)
        target_link_libraries(liberty-vehicle-editor PRIVATE comdlg32)
        if(MSVC)
            target_sources(liberty-vehicle-editor PRIVATE utf8.manifest)
        endif()
    else()
        set_target_properties(liberty-vehicle-editor PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
        get_target_property(glfw_imported glfw IMPORTED)
        get_target_property(glfw_type glfw TYPE)
        if(glfw_imported AND glfw_type STREQUAL "SHARED_LIBRARY")
            install(IMPORTED_RUNTIME_ARTIFACTS glfw LIBRARY DESTINATION lib)
        endif()
    endif()
    install(TARGETS liberty-vehicle-editor RUNTIME DESTINATION .)
    if(WIN32 AND BUILD_SHARED_LIBS)
        install(FILES $<TARGET_RUNTIME_DLLS:liberty-vehicle-editor> DESTINATION .)
    endif()
    install(FILES README.md NOTICE.md ../../LICENSE DESTINATION .)
    if(VEHICLE_EDITOR_MANAGER_RUNTIME)
        install(DIRECTORY "${VEHICLE_EDITOR_MANAGER_RUNTIME}/" DESTINATION manager USE_SOURCE_PERMISSIONS)
    endif()
    install(FILES ../../librw/LICENSE DESTINATION . RENAME LIBRW-LICENSE)
    install(FILES ../../librw/skeleton/imgui/LICENSE_imgui.txt ../../librw/skeleton/imgui/LICENSE_imguizmo.txt DESTINATION .)
    install(FILES GLFW-LICENSE.md DESTINATION .)
    set(CPACK_GENERATOR ZIP)
    set(CPACK_PACKAGE_FILE_NAME "LibertyVehicleEditor-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}")
    include(CPack)
endif()
