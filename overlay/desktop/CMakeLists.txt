endif()

    message(FATAL_ERROR "This bring-up target requires Windows or Linux x64.")
endif()

set(CMAKE_CXX_EXTENSIONS OFF)

endif()

  find_library(FFMPEG_${component} NAMES ${component} HINTS "${LIBERTYVR_DESKTOP_PREFIX}/lib" "${LIBERTYVR_DESKTOP_PREFIX}/bin" REQUIRED)
  target_link_libraries(libertyvr-media INTERFACE "${FFMPEG_${component}}")

if(NOT TARGET OpenXR::openxr_loader)
    # Source contains generated loader files; Python is not required.
    # FETCHCONTENT_SOURCE_DIR_OPENXR_SDK supports an existing offline checkout.
    include(FetchContent)
    set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
    set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
    set(BUILD_LOADER ON CACHE BOOL "" FORCE)
    set(DYNAMIC_LOADER ON CACHE BOOL "" FORCE)
    FetchContent_Declare(openxr_sdk
        URL https://github.com/KhronosGroup/OpenXR-SDK/archive/refs/tags/release-1.1.58.zip
        URL_HASH SHA256=43d75ce00e31be8f662b58e8c74af8c04156bcaa7acbfc19112aa2880f3a3669)
    FetchContent_MakeAvailable(openxr_sdk)
endif()
get_filename_component(LIBERTYVR_OVERLAY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(LIBRW_DIR "${LIBERTYVR_OVERLAY_DIR}/../librw" CACHE PATH "Bundled librw source")
if(NOT EXISTS "${LIBRW_DIR}/rw.h")
    # Also support a manager-assembled source tree.
    set(LIBRW_DIR "${LIBERTYVR_OVERLAY_DIR}/vendor/librw")

include("${LIBERTYVR_OVERLAY_DIR}/cmake/LibertyVulkan.cmake")
set(SHADER_OUTPUTS "")
foreach(shader "mirror.vert" "mirror.frag" "desktop_ui.vert" "desktop_ui.frag" "desktop_overlay.frag")
    get_filename_component(stage "${shader}" LAST_EXT)
    if(stage STREQUAL ".vert")
        set(stage vertex)
    else()
        set(stage fragment)
    endif()
    string(REPLACE "." "_" symbol "${shader}")
    if(symbol MATCHES "^mirror")
        set(symbol "desktop_${symbol}")
    endif()
    compile_shader("${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}" ${stage} ${symbol}
        "${SPIRV_OUT_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders")
endforeach()
add_custom_target(desktop_shaders DEPENDS ${SHADER_OUTPUTS})
set(DESKTOP_IMGUI "${LIBRW_DIR}/skeleton/imgui")
add_library(libertyvr-desktop STATIC DesktopWindow.cpp DesktopGpu.cpp DesktopAudio.cpp
    "${DESKTOP_IMGUI}/imgui.cpp" "${DESKTOP_IMGUI}/imgui_draw.cpp" "${DESKTOP_IMGUI}/imgui_demo.cpp")
add_dependencies(libertyvr-desktop desktop_shaders)
target_include_directories(libertyvr-desktop PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${LIBERTYVR_OVERLAY_DIR}/src/xr" "${LIBERTYVR_OVERLAY_DIR}/src/core" PRIVATE "${DESKTOP_IMGUI}" "${SPIRV_OUT_DIR}" "${LIBERTYVR_OVERLAY_DIR}/src/audio" "${LIBERTYVR_OVERLAY_DIR}/android/app/src/main/cpp")
target_compile_definitions(libertyvr-desktop PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_link_libraries(libertyvr-desktop PUBLIC Vulkan::Vulkan SDL3::SDL3 libertyvr-media Threads::Threads)
if(LIBERTYVR_RE3_ROOT)
    set(RE3_ROOT "${LIBERTYVR_RE3_ROOT}")
    set(LIBERTYVR_APP_DIR "${LIBERTYVR_OVERLAY_DIR}/android/app/src/main/cpp")
    set(LIBERTYVR_SOURCES game_main.cpp platform_desktop.cpp platform_native.cpp profile_posix.cpp renderdoc_desktop.cpp streaming_desktop.cpp DesktopGameInput.cpp
        "${LIBERTYVR_APP_DIR}/android_main.cpp" "${LIBERTYVR_APP_DIR}/xr_vulkan_session.cpp")
    include("${LIBERTYVR_OVERLAY_DIR}/cmake/LibertyGameSources.cmake")
    foreach(game_target libertyvr-game libertyvr-upstate libertyvr-render-debug libertyvr-render-debug-upstate)
        set(target_sources ${LIBERTYVR_SOURCES})
        if(game_target MATCHES "^libertyvr-render-debug")
            list(REMOVE_ITEM target_sources game_main.cpp "${LIBERTYVR_APP_DIR}/xr_vulkan_session.cpp")
            list(APPEND target_sources render_debug_main.cpp RenderDebugClient.cpp DebugWorld.cpp DebugVulkan.cpp DebugSessionServices.cpp)
        endif()
        add_executable(${game_target} ${target_sources})
        add_dependencies(${game_target} libertyvr_shaders)
        target_include_directories(${game_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
            "${LIBERTYVR_OVERLAY_DIR}/src/audio" "${LIBERTYVR_OVERLAY_DIR}/src/xr"
            "${LIBERTYVR_OVERLAY_DIR}/src/core" "${LIBERTYVR_APP_DIR}" "${SPIRV_OUT_DIR}" ${GAME_INCLUDES})
        target_compile_definitions(${game_target} PRIVATE LIBRW RW_VULKAN GTA_VR_WEAPONS
            AUDIO_ANDROID LIBERTYVR_DESKTOP XR_USE_GRAPHICS_API_VULKAN
            LIBERTYVR_DEV_TOOLS=$<BOOL:${LIBERTYVR_DEV_TOOLS}>
            LIBERTYVR_VEHICLE_CAPTURE=0 CMAKE_NO_AUTOLINK NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
        if(MSVC)
            target_link_libraries(${game_target} PRIVATE dbghelp shell32)
            target_include_directories(${game_target} PRIVATE "${LIBERTYVR_OVERLAY_DIR}/windows/compat")
            target_compile_definitions(${game_target} PRIVATE WITHWINDOWS)
            target_compile_options(${game_target} PRIVATE "/FI${LIBERTYVR_OVERLAY_DIR}/windows/compat/RuntimeCompat.h" /bigobj /MP8 /Zi)
            target_link_options(${game_target} PRIVATE /DEBUG)
        else()
            target_compile_options(${game_target} PRIVATE -g -Wno-invalid-offsetof)
            target_link_libraries(${game_target} PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
        endif()
        target_link_libraries(${game_target} PRIVATE librw libertyvr-desktop)
        if(game_target MATCHES "^libertyvr-render-debug")
            target_compile_definitions(${game_target} PRIVATE LIBERTYVR_RENDER_DEBUG=1)
            target_include_directories(${game_target} PRIVATE "${DESKTOP_IMGUI}")
            # Headers are used for shared stereo math only. Never link the loader.
            target_link_libraries(${game_target} PRIVATE OpenXR::headers)
        else()
            target_link_libraries(${game_target} PRIVATE OpenXR::openxr_loader)
        endif()
        if(TARGET openxr_loader AND NOT game_target MATCHES "^libertyvr-render-debug")
            add_custom_command(TARGET ${game_target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E
                copy_if_different $<TARGET_FILE:openxr_loader> $<TARGET_FILE_DIR:${game_target}>)
        endif()
        if(game_target STREQUAL "libertyvr-upstate" OR game_target STREQUAL "libertyvr-render-debug-upstate")
            target_compile_definitions(${game_target} PRIVATE LCVR_UPSTATE_ENGINE=1)
        endif()
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pcvr-mod-capabilities.txt" "upstate-liberty/1\n")
endif()
